Iniciar trayecto:

        Body: {"startlat":"..", "startlon":"..", "destinationlat": ".." "destinationlon": "..", "route": "(Ruta generada entre inicio y destino)",state": "iniciado"}

        Body: {"pin": "1234", "status": "completed", "destinationlat": "..", "destinationlon": ".."}
